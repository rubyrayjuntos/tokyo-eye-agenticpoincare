🧬 Structural Biology & Modeling Logic
The Conductance Formula
Python

Implication: Because you are dividing by the exponentiated sum of depths, buried residues (high cone depth) will have tiny conductances (massive resistance). Conversely, exposed surface residues (low cone depth) will have huge conductances (low resistance).

Does this match your intent? In your comments, you state: "Lower effective resistance = stronger allosteric coupling." If buried residues have massive resistance, energy/information will prefer to route through the fluid, highly exposed surface loops rather than the rigid hydrophobic core. If your goal is to model allosteric propagation through dense, rigid structural cores, your formula should make conductance proportional to depth, not inversely proportional.

import numpy as np
import scipy.linalg as sla
import networkx as nx
from scipy.spatial import KDTree

def _build_conductance_graph(
    ca_coords: np.ndarray,
    cone_depths: np.ndarray,
    cutoff: float,
) -> nx.Graph:
    G = nx.Graph()
    n = len(ca_coords)
    
    # Track which nodes actually have valid structural data
    valid_nodes = [i for i in range(n) if not np.isnan(ca_coords[i]).any()]
    
    for i in valid_nodes:
        G.add_node(i, xyz=ca_coords[i].tolist())

    if not valid_nodes:
        return G

    # Build KDTree only on valid spatial coordinates
    valid_coords = ca_coords[valid_nodes]
    tree = KDTree(valid_coords)
    pairs = tree.query_pairs(cutoff)
    
    for pi, pj in pairs:
        i, j = valid_nodes[pi], valid_nodes[pj]
        # Ensure depth values don't overflow/underflow exponentially
        # ADJUST THIS FORMULA based on whether core or surface should conduct
        conductance = float(1.0 / np.exp(cone_depths[i] + cone_depths[j]))
        G.add_edge(i, j, conductance=conductance)

    return G


def _spectral_analysis(G: nx.Graph) -> dict:
    if G.number_of_nodes() < 3 or not nx.is_connected(G):
        # Handle disconnected components gracefully
        return {"lambda_2": 0.0, "hinge_residues": [], "fiedler_vector": []}

    # Use dense solver safely only if graph is small, or swap to scipy.sparse.linalg.eigsh
    L = nx.laplacian_matrix(G, weight="conductance").toarray().astype(np.float64)
    eigenvalues, eigenvectors = sla.eigh(L)

    lambda_2 = float(eigenvalues[1])
    fiedler = eigenvectors[:, 1]

    # Hinge residues: Defined as boundary nodes where Fiedler value crosses near 0
    # or identify structural partitions using the sign vector
    hinges = []
    epsilon = 0.02  # Define a tolerance band around the partition boundary
    for idx, node_id in enumerate(G.nodes()):
        if abs(fiedler[idx]) < epsilon:
            hinges.append(int(node_id))

    return {
        "lambda_2": lambda_2,
        "hinge_residues": hinges,
        "fiedler_vector": fiedler.tolist(),
    }
